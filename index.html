-- ============================================================ -- יאלה - עדכון RLS לתמיכה בבקשות הצטרפות מלאות -- ============================================================ -- הרץ פעם אחת ב-Supabase SQL Editor -- מתקן: מארגן יכול לאשר/לדחות בקשות, כולם רואים מי אושר -- ============================================================ -- מחק policies קיימים על join_requests DROP POLICY IF EXISTS "View own requests" ON public.join_requests; DROP POLICY IF EXISTS "Create own requests" ON public.join_requests; DROP POLICY IF EXISTS "j1" ON public.join_requests; DROP POLICY IF EXISTS "j2" ON public.join_requests; DROP POLICY IF EXISTS "Users can view requests they made or for plans they created" ON public.join_requests; DROP POLICY IF EXISTS "Authenticated users can create join requests" ON public.join_requests; DROP POLICY IF EXISTS "Users can cancel own requests, creators can approve/decline" ON public.join_requests; DROP POLICY IF EXISTS "anyone_can_view_requests" ON public.join_requests; DROP POLICY IF EXISTS "auth_users_can_insert_own_requests" ON public.join_requests; DROP POLICY IF EXISTS "users_can_update_own_or_plan_creators_can" ON public.join_requests; -- צור policies חדשים נכונים -- 1. כל אחד יכול לראות בקשות (אחרת אי אפשר להציג מי הצטרף) CREATE POLICY "anyone_can_view_requests" ON public.join_requests FOR SELECT USING (true); -- 2. משתמש מחובר יכול לשלוח בקשה רק עבור עצמו CREATE POLICY "auth_users_can_insert_own_requests" ON public.join_requests FOR INSERT WITH CHECK (auth.uid() = user_id); -- 3. עדכון - המבקש יכול לבטל, יוצר התוכנית יכול לאשר/לדחות CREATE POLICY "users_can_update_own_or_plan_creators_can" ON public.join_requests FOR UPDATE USING ( auth.uid() = user_id OR auth.uid() IN (SELECT creator_id FROM public.plans WHERE id = plan_id) ); -- ============================================================ -- אם הרצת זה - אמור להופיע "Success. No rows returned" -- ============================================================
